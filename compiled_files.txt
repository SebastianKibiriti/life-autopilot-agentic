91: class RoutesEstimator:
92:     """Estimates walking travel duration and distance with Google Routes API or fallback."""
93: 
94:     def __init__(self, api_key: str | None = None) -> None:
95:         self.api_key = api_key or os.getenv("ROUTES_API_KEY")
96: 
97:     def estimate_walking(
98:         self, origin: Location, destination: Destination
99:     ) -> TravelEstimate:
100:         # 1. If API Key present, try Google Routes API
101:         if self.api_key:
102:             try:
103:                 with httpx.Client(timeout=5.0) as client:
104:                     resp = client.post(
105:                         "https://routes.googleapis.com/directions/v2:computeRoutes",
106:                         headers={
107:                             "Content-Type": "application/json",
108:                             "X-Goog-Api-Key": self.api_key,
109:                             "X-Goog-FieldMask": "routes.duration,routes.distanceMeters",
110:                         },
111:                         json={
112:                             "origin": {
113:                                 "location": {
114:                                     "latLng": {
115:                                         "latitude": origin.latitude,
116:                                         "longitude": origin.longitude,
117:                                     }
118:                                 }
119:                             },
120:                             "destination": {
121:                                 "location": {
122:                                     "latLng": {
123:                                         "latitude": destination.latitude,
124:                                         "longitude": destination.longitude,
125:                                     }
126:                                 }
127:                             },
128:                             "travelMode": "WALK",
129:                         },
130:                     )
131:                     if resp.status_code == 200:
132:                         data = resp.json()
133:                         routes = data.get("routes", [])
134:                         if routes:
135:                             route = routes[0]
136:                             dist_meters = route.get("distanceMeters", 0)
137:                             dur_str = route.get("duration", "0s").rstrip("s")
138:                             dur_seconds = int(float(dur_str))
139:                             dur_minutes = max(1, math.ceil(dur_seconds / 60))
140:                             return TravelEstimate(
141:                                 origin=origin,
142:                                 destination=destination,
143:                                 mode="walking",
144:                                 distance_meters=dist_meters,
145:                                 duration_seconds=dur_seconds,
146:                                 duration_minutes=dur_minutes,
147:                                 provider=RouteProvider.ROUTES,
148:                                 estimated_at=datetime.now(timezone.utc),
149:                             )
150:             except Exception:
151:                 pass
152: 
153:         # 2. High accuracy Haversine walking calculation fallback
154:         lat1, lon1 = origin.latitude, origin.longitude
155:         lat2, lon2 = destination.latitude, destination.longitude
156:         r = 6371000  # Earth radius in meters
157: 
158:         phi1 = math.radians(lat1)
159:         phi2 = math.radians(lat2)
160:         delta_phi = math.radians(lat2 - lat1)
161:         delta_lambda = math.radians(lon2 - lon1)
162: 
163:         a = (
164:             math.sin(delta_phi / 2.0) ** 2
165:             + math.cos(phi1) * math.cos(phi2) * math.sin(delta_lambda / 2.0) ** 2
166:         )
167:         c = 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
168:         direct_distance = r * c
169: 
170:         # Average pedestrian walking speed: ~1.35 m/s (~4.86 km/h)
171:         # Factor in urban pedestrian path network factor ~1.25
172:         pedestrian_distance = int(direct_distance * 1.25)
173:         walking_speed_mps = 1.35
174:         duration_seconds = int(pedestrian_distance / walking_speed_mps)
175:         duration_minutes = max(1, math.ceil(duration_seconds / 60))
176: 
177:         return TravelEstimate(
178:             origin=origin,
179:             destination=destination,
180:             mode="walking",
181:             distance_meters=pedestrian_distance,
182:             duration_seconds=duration_seconds,
183:             duration_minutes=duration_minutes,
184:             provider=RouteProvider.FALLBACK,
185:             estimated_at=datetime.now(timezone.utc),
186:         )
